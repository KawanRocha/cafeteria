Não estiver perto do cartão * 2. Estiver suficientemente distante * da posição anterior */ if ( !estaPertoDoCartao && distanciaAtual > 120 ) { posicaoValida = true; break; } } /* * Caso não encontre uma posição depois * de várias tentativas, usa uma posição * segura nos cantos da tela. */ if (!posicaoValida) { const possibilidades = [ { x: margemTela, y: margemTela }, { x: larguraTela - larguraBotao - margemTela, y: margemTela }, { x: margemTela, y: alturaTela - alturaBotao - margemTela }, { x: larguraTela - larguraBotao - margemTela, y: alturaTela - alturaBotao - margemTela } ]; // Escolhe um dos cantos aleatoriamente const escolha = possibilidades[ Math.floor( Math.random() * possibilidades.length ) ]; novoX = escolha.x; novoY = escolha.y; } /* * Garante que nunca saia da tela. */ novoX = Math.max( margemTela, Math.min( novoX, larguraTela - larguraBotao - margemTela ) ); novoY = Math.max( margemTela, Math.min( novoY, alturaTela - alturaBotao - margemTela ) ); /* * Move o botão. */ btnNao.style.left = novoX + "px"; btnNao.style.top = novoY + "px"; /* * Troca a frase. */ btnNao.innerText = frasesNao[indiceFrase]; indiceFrase++; if ( indiceFrase >= frasesNao.length ) { indiceFrase = 0; } }
